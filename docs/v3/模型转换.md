在我们训练生成.pt模型后还需要进行两步
1. 将.pt转换为ONNX
```
conda activate yolov8-rk-export
```
2. 将ONNX转换为.rknn
```
conda activate python3.8-tk2-2.0
conda deactivate

  python convert_yolov8_rknn.py \
  --onnx ~/work/yolov8_rk3588/models/best.onnx \
  --output ~/work/yolov8_rk3588/models/best_fp.rknn \
  --dataset ~/work/yolov8_rk3588/calib/dataset.txt \
  --verbose \
  2>&1 | tee ~/work/yolov8_rk3588/logs/convert_fp.log

    python convert_yolov8_rknn.py \
  --onnx ~/work/yolov8_rk3588/models/best.onnx \
  --output ~/work/yolov8_rk3588/models/best_i8.rknn \
  --dataset ~/work/yolov8_rk3588/calib/dataset.txt \
  --quant \
  --verbose \
  2>&1 | tee ~/work/yolov8_rk3588/logs/convert_i8.log
```
